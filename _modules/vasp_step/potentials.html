    """The potential of each element: the one chosen, else the variant's,
    else the set's default.

    Parameters
    ----------
    potential_set : str
        e.g. "potpaw_PBE.64".
    elements : [str]
        Element symbols.
    chosen : {str: str} or None
        Potentials chosen explicitly, by element.
    variant : str or None
        A named variant, e.g. "hard".

    Returns
    -------
    {str: str}
    """
    chosen = chosen or {}
    overlay = variants[variant] if variant else {}
    defaults = default_potentials.get(potential_set, {})
    result = {}
    for element in elements:
        if element in chosen:
            result[element] = chosen[element]
        elif element in overlay:
            result[element] = overlay[element]
        elif element in defaults:
            result[element] = defaults[element]
        else:
            raise ValueError(
                f"No potential for {element} in the set {potential_set}: choose "
                "one explicitly."
            )
    return result
